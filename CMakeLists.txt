cmake_minimum_required(VERSION 3.16)

project(RetroDisc)


# ================================================================
# C++
# ================================================================

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)


# ================================================================
# DEPENDENCIES
# ================================================================

find_package(
    nlohmann_json
    REQUIRED
)


# ================================================================
# RUNTIME CONFIGURATIONS
#
# Every JSON file in:
#
#     runtime/*.json
#
# is embedded automatically.
#
# Runtime names are NOT hardcoded.
# ================================================================

file(
    GLOB
    RETRODISC_RUNTIME_CONFIG_FILES
    CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/runtime/*.json"
)


if(NOT RETRODISC_RUNTIME_CONFIG_FILES)

    message(
        FATAL_ERROR
        "No runtime configuration files found in: ${CMAKE_SOURCE_DIR}/runtime"
    )

endif()


# ================================================================
# GENERATED DIRECTORY
# ================================================================

set(
    RETRODISC_GENERATED_DIR
    "${CMAKE_BINARY_DIR}/generated"
)


file(
    MAKE_DIRECTORY
    "${RETRODISC_GENERATED_DIR}"
)


set(
    RETRODISC_GENERATED_RUNTIME_CONFIG_CPP
    "${RETRODISC_GENERATED_DIR}/embedded_runtime_config.cpp"
)


# ================================================================
# START GENERATED C++ FILE
# ================================================================

file(
    WRITE
    "${RETRODISC_GENERATED_RUNTIME_CONFIG_CPP}"
    "/*\n"
    " * THIS FILE IS GENERATED BY CMAKE.\n"
    " * DO NOT EDIT.\n"
    " */\n\n"
    "#include \"runtime_config.hpp\"\n\n"
    "#include <string>\n"
    "#include <unordered_map>\n\n"
    "namespace\n"
    "{\n\n"
    "const std::unordered_map<std::string, std::string>& runtimeConfigs()\n"
    "{\n"
    "    static const std::unordered_map<std::string, std::string> configs =\n"
    "    {\n"
)


# ================================================================
# EMBED EVERY RUNTIME JSON
# ================================================================

foreach(
    RUNTIME_CONFIG
    IN LISTS RETRODISC_RUNTIME_CONFIG_FILES
)

    # ------------------------------------------------------------
    # Runtime name
    #
    # Example:
    #
    # runtime/foo.json
    #
    # becomes:
    #
    # foo
    # ------------------------------------------------------------

    get_filename_component(
        RUNTIME_NAME
        "${RUNTIME_CONFIG}"
        NAME_WE
    )


    # ------------------------------------------------------------
    # Read JSON
    # ------------------------------------------------------------

    file(
        READ
        "${RUNTIME_CONFIG}"
        RUNTIME_CONTENT
    )


    # ------------------------------------------------------------
    # Escape backslashes
    # ------------------------------------------------------------

    string(
        REPLACE
        "\\"
        "\\\\"
        RUNTIME_CONTENT
        "${RUNTIME_CONTENT}"
    )


    # ------------------------------------------------------------
    # Escape double quotes
    # ------------------------------------------------------------

    string(
        REPLACE
        "\""
        "\\\""
        RUNTIME_CONTENT
        "${RUNTIME_CONTENT}"
    )


    # ------------------------------------------------------------
    # Escape carriage returns
    # ------------------------------------------------------------

    string(
        REPLACE
        "\r"
        "\\r"
        RUNTIME_CONTENT
        "${RUNTIME_CONTENT}"
    )


    # ------------------------------------------------------------
    # Escape newlines
    # ------------------------------------------------------------

    string(
        REPLACE
        "\n"
        "\\n"
        RUNTIME_CONTENT
        "${RUNTIME_CONTENT}"
    )


    # ------------------------------------------------------------
    # Append generated entry
    # ------------------------------------------------------------

    file(
        APPEND
        "${RETRODISC_GENERATED_RUNTIME_CONFIG_CPP}"
        "        {\n"
        "            \"${RUNTIME_NAME}\",\n"
        "            \"${RUNTIME_CONTENT}\"\n"
        "        },\n"
    )

endforeach()


# ================================================================
# FINISH GENERATED C++ FILE
# ================================================================

file(
    APPEND
    "${RETRODISC_GENERATED_RUNTIME_CONFIG_CPP}"
    "    };\n\n"
    "    return configs;\n"
    "}\n\n"
    "}\n\n"
    "bool getEmbeddedRuntimeConfig(\n"
    "    const std::string& runtime,\n"
    "    std::string& output\n"
    ")\n"
    "{\n"
    "    const auto& configs = runtimeConfigs();\n\n"
    "    const auto it = configs.find(runtime);\n\n"
    "    if(it == configs.end())\n"
    "    {\n"
    "        return false;\n"
    "    }\n\n"
    "    output = it->second;\n\n"
    "    return true;\n"
    "}\n"
)


# ================================================================
# EXECUTABLE
# ================================================================

add_executable(
    RetroDisc

    src/main.cpp

    src/config/path.cpp
    src/config/create_config.cpp
    src/config/manifest.cpp
    src/config/load.cpp

    src/filesystem/prepare.cpp
    src/filesystem/overlay.cpp
    src/filesystem/cleanup.cpp
    src/filesystem/mount.cpp
    src/filesystem/directory.cpp
    src/filesystem/command.cpp

    src/registry/validation.cpp
    src/registry/copy.cpp
    src/registry/prefix.cpp

    src/runtime.cpp
    src/discord.cpp

    "${RETRODISC_GENERATED_RUNTIME_CONFIG_CPP}"
)


# ================================================================
# INCLUDE DIRECTORIES
# ================================================================

target_include_directories(
    RetroDisc

    PRIVATE

    "${CMAKE_SOURCE_DIR}/include"
    "${RETRODISC_GENERATED_DIR}"
)


# ================================================================
# LIBRARIES
# ================================================================

target_link_libraries(
    RetroDisc

    PRIVATE

    nlohmann_json::nlohmann_json
)
